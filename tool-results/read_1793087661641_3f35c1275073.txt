    68→/**
    69→ * Game scene — owns the canvas gameplay.
    70→ *
    71→ * INPUT MODEL:
    72→ *   - The universal Answer Pad (React/DOM) is always visible during play on
    73→ *     every platform. A tap emits bridge "cast" {digit}.
    74→ *   - PC keyboard 0-9/numpad casts directly (desktop only). Esc pauses.
    75→ *   - Every input is a definite digit 0-9 — there is no "miscast" concept.

    78→ *
    79→ * TEN ELEMENT SPELL EFFECTS: on a correct kill, castFx(digit, x, y) dispatches

    81→ *
    82→ * PACING: per-level (see LEVELS in tuning.ts).

    84→ * QUESTION MODEL (per-level): each wisp carries its own fixed question
    85→ * generated by generateQuestion() based on the active level's operators.
    86→ * Answer = last digit of the result. Uniqueness re-roll (max 20 tries) so

   306→  private refreshKeepFrame(): void {
   307→    if (!this.keepImg) return;
   308→    // Only setFrame if the texture has multiple frames (procedural keep has 2).
   309→    // The PNG castle is a single-frame image (frameTotal=1) — no frame switching.
   310→    const tex = this.keepImg.texture;
   311→    if (tex && tex.frameTotal > 2) {
   312→      const cracked = this.wallHP <= TUNING.KEEP_CRACK_HP_THRESHOLD;
   313→      this.keepImg.setFrame(cracked ? 1 : 0);

   317→  /* ----------------------------- update ------------------------------ */
   318→
   319→  update(time: number, deltaMs: number): void {
   320→    if (!this.running || this.paused) return;
   321→
   322→    const cap = this.airborneCap();
   323→    this.spawnTimer -= deltaMs;
   324→    if (this.spawnTimer <= 0 && this.wisps.length < cap) {
   325→      this.spawnWisp();
   326→      this.spawnTimer += this.spawnInterval;
   327→      if (this.spawnTimer < 0) this.spawnTimer = this.spawnInterval;
   328→    }
   329→    if (
   330→      this.goSecondSpawnPending &&
   331→      this.wisps.length < cap &&
   332→      time >= this.runStartTime + TUNING.GO_DOUBLE_SPAWN_DELAY
   333→    ) {
   334→      this.goSecondSpawnPending = false;
   335→      this.spawnWisp();
   336→    }
   337→
   338→    for (const w of this.wisps) {
   339→      w.container.y += w.speedPerMs * deltaMs;
   340→      w.container.x =
   341→        w.baseX +
   342→        Math.sin(time * TUNING.SWAY_FREQ_PER_MS + w.phase) *
   343→          TUNING.SWAY_AMPLITUDE;
   344→      // Animation: if the enemy is in death state and animDone, destroy and remove
   345→      if (w.animState === "death" && w.animDone) {
   346→        const di = this.wisps.indexOf(w);
   347→        if (di >= 0) this.wisps.splice(di, 1);
   348→        w.container.destroy();
   349→        continue;
   350→      }
   351→      // If in attack state and animDone, process wall hit
   352→      if (w.animState === "attack" && w.animDone) {
   353→        // Attack animation complete; now apply wall damage
   354→        this.wallHit(w);
   355→        continue;
   356→      }
   357→      // Skip normal movement for death/attack states
   358→      if (w.animState === "death" || w.animState === "attack") continue;
   359→
   360→      // Normal movement (enter + idle states)
   361→      w.container.y += w.speedPerMs * deltaMs;
   362→      w.container.x =
   363→        w.baseX +
   364→        Math.sin(time * TUNING.SWAY_FREQ_PER_MS + w.phase) *
   365→          TUNING.SWAY_AMPLITUDE;
   366→
   367→      // Frame animation: only for procedural fallback textures (no Phaser anim).
   368→      // PNG spritesheets use the Phaser animation system (playWispAnim).
   369→      const tex = w.sprite.texture;
   370→      const animKey = `${tex.key}_idle`;
   371→      if (w.animState === "idle" && !this.anims.exists(animKey) && tex && tex.frameTotal > 1) {
   372→        // Procedural fallback: manual flutter
   373→        w.sprite.setFrame(Math.floor(time / 140) % 4);
   374→      }
   375→
   376→      w.trailTimer += deltaMs;
   377→      if (w.trailTimer >= TUNING.WISP_TRAIL_INTERVAL_MS) {
   378→        w.trailTimer = 0;
   379→        this.spawnWispTrail(w);
   380→      }
   381→      if (w.container.y >= TUNING.WALL_CONTACT_Y) {
   382→        // Switch to attack state — the wall hit logic runs when attack anim completes
   383→        w.animState = "attack";
   384→        w.animDone = false;
   385→        this.playWispAnim(w, "attack");
   386→        // If no attack animation exists (fallback), resolve immediately
   387→        if (!this.anims.exists(`${w.sprite.texture.key}_attack`)) {
   388→          this.wallHit(w);
   389→        }
   390→      }
   391→    }
   392→  }

   394→  private airborneCap(): number {
   395→    const cfg = this.levelCfg;
   396→    for (const phase of cfg.airborne) {
   397→      if (this.correctCount < phase.until) return phase.cap;
   398→    }
   399→    return cfg.airborne[cfg.airborne.length - 1].cap;
   400→  }

   404→  /** Register Phaser animations for each enemy type's enter/idle/attack/death states. */
   405→  private registerEnemyAnimations(): void {
   406→    for (const key of ENEMY_KEYS) {
   407→      const ranges = ENEMY_FRAME_RANGES[key];
   408→      if (!ranges) continue;
   409→      // Only register if the texture exists and has enough frames
   410→      if (!this.textures.exists(key)) continue;
   411→      const tex = this.textures.get(key);
   412→      const totalFrames = tex.frameTotal - 1; // subtract __BASE
   413→      if (totalFrames < ranges[7] + 1) continue; // not enough frames
   414→
   415→      const [eS, eE, iS, iE, aS, aE, dS, dE] = ranges;
   416→      // Enter: play once, then transition to idle
   417→      this.anims.create({
   418→        key: `${key}_enter`,
   419→        frames: this.anims.generateFrameNumbers(key, { start: eS, end: eE }),
   420→        frameRate: 10,
   421→        repeat: 0,
   422→      });
   423→      // Idle: loop
   424→      this.anims.create({
   425→        key: `${key}_idle`,
   426→        frames: this.anims.generateFrameNumbers(key, { start: iS, end: iE }),
   427→        frameRate: 4,
   428→        repeat: -1,
   429→      });
   430→      // Attack: play once
   431→      this.anims.create({
   432→        key: `${key}_attack`,
   433→        frames: this.anims.generateFrameNumbers(key, { start: aS, end: aE }),
   434→        frameRate: 8,
   435→        repeat: 0,
   436→      });
   437→      // Death: play once, then mark done
   438→      this.anims.create({
   439→        key: `${key}_death`,
   440→        frames: this.anims.generateFrameNumbers(key, { start: dS, end: dE }),
   441→        frameRate: 8,
   442→        repeat: 0,
   443→      });
   444→    }
   445→  }

   447→  /** Start an animation on a wisp sprite, handling fallback to procedural textures. */
   448→  private playWispAnim(w: Wisp, state: "enter" | "idle" | "attack" | "death"): void {
   449→    const animKey = `${w.sprite.texture.key}_${state}`;
   450→    if (this.anims.exists(animKey)) {
   451→      // Phaser Image doesn't have .play() — use the Sprite-based approach
   452→      // by stopping any current anim and using the texture's frames directly
   453→      // via the animation manager
   454→      const anim = this.anims.get(animKey);
   455→      if (anim) {
   456→        // Manual frame stepping for Image objects (which don't have .play())
   457→        let frameIdx = anim.frames[0].frame;
   458→        w.sprite.setFrame(frameIdx);
   459→        // Set up a timer-based frame cycle
   460→        const frames = anim.frames;
   461→        let currentFrame = 0;
   462→        const frameRate = anim.frameRate || 4;
   463→        const frameMs = 1000 / frameRate;
   464→        // Clear any existing timer for this wisp
   465→        // Use a simple recursive delayedCall approach
   466→        const stepFrame = () => {
   467→          if (w.animState !== state) return; // state changed, stop stepping
   468→          currentFrame = (currentFrame + 1) % frames.length;
   469→          if (currentFrame === 0 && anim.repeat === 0) {
   470→            // Non-repeating animation completed
   471→            if (state === "enter") {
   472→              w.animState = "idle";
   473→              this.playWispAnim(w, "idle");
   474→            } else if (state === "death") {
   475→              w.animDone = true;
   476→            } else if (state === "attack") {
   477→              w.animDone = true;
   478→            }
   479→            return;
   480→          }
   481→          w.sprite.setFrame(frames[currentFrame].frame);
   482→          this.time.delayedCall(frameMs, stepFrame);
   483→        };
   484→        this.time.delayedCall(frameMs, stepFrame);
   485→      }
   486→    } else {
   487→      // Fallback: procedural textures had 4 flutter frames; just set frame 0
   488→      w.sprite.setFrame(0);
   489→    }
   490→  }

   492→  private spawnWisp(): void {
   493→    const q = this.generateQuestion();
   494→
   495→    const fallTime = this.levelCfg.fallStart;
   496→    const distance = TUNING.WALL_CONTACT_Y - TUNING.WISP_SPAWN_Y;
   497→    const speedPerMs = distance / fallTime;
   498→
   499→    const baseX = Phaser.Math.Between(
   500→      TUNING.WISP_SPAWN_X_MIN,
   501→      TUNING.WISP_SPAWN_X_MAX,
   502→    );
   503→    const container = this.add.container(baseX, TUNING.WISP_SPAWN_Y);
   504→    container.setDepth(3);
   505→
   506→    // Pick one of 4 enemy variants at random; use PNG texture if loaded,
   507→    // otherwise fall back to procedural wisp texture.
   508→    const idx = Phaser.Math.Between(0, ENEMY_KEYS.length - 1);
   509→    const pngKey = ENEMY_KEYS[idx];

   548→  /**
   549→   * Generate one question for the active level, re-rolling (max 20 tries)
   550→   * until its answer differs from every currently airborne wisp's answer.
   551→   */
   552→  private generateQuestion(): { question: string; answer: number } {
   553→    const cfg = this.levelCfg;
   554→    const airborne = new Set(this.wisps.map((w) => w.answer));
   555→    for (let i = 0; i < TUNING.DIGIT_PICK_TRIES; i++) {
   556→      const q = this.generateOneQuestion(cfg.ops);
   557→      if (!airborne.has(q.answer)) return q;
   558→    }
   559→    return this.generateOneQuestion(cfg.ops);
   560→  }
   561→
   562→  private generateOneQuestion(ops: Op[]): { question: string; answer: number } {
   563→    const op = ops[Phaser.Math.Between(0, ops.length - 1)];
   564→    const D = () => Phaser.Math.Between(TUNING.DIGIT_MIN, TUNING.DIGIT_MAX);
   565→
   566→    if (op === "+") {
   567→      if (this.level === "veryEasy") {
   568→        let a = D(), b = D();
   569→        for (let i = 0; i < 20 && a + b > 9; i++) { a = D(); b = D(); }
   570→        if (a + b > 9) { a = Math.min(a, 4); b = Math.min(b, 9 - a); }
   571→        return { question: `${a} + ${b}`, answer: a + b };
   572→      }
   573→      const a = D(), b = D();
   574→      return { question: `${a} + ${b}`, answer: (a + b) % 10 };
   575→    }
   576→
   577→    if (op === "-") {
   578→      let a = D(), b = D();
   579→      if (a < b) { const t = a; a = b; b = t; }
   580→      return { question: `${a} - ${b}`, answer: a - b };
   581→    }
   582→
   583→    if (op === "x") {
   584→      const a = D(), b = D();
   585→      return { question: `${a} x ${b}`, answer: (a * b) % 10 };
   586→    }
   587→
   588→    // divide: pick b in 2-9, q in 1-9, a = b*q. Answer = q (last digit).
   589→    const b = Phaser.Math.Between(2, 9);
   590→    const q = Phaser.Math.Between(1, 9);
   591→    const a = b * q;
   592→    return { question: `${a} / ${b}`, answer: q % 10 };
   593→  }

   595→  private getLowest(): Wisp | null {
   596→    if (this.wisps.length === 0) return null;
   597→    let best = this.wisps[0];
   598→    for (const w of this.wisps) {
   599→      if (w.container.y > best.container.y) best = w;
   600→    }
   601→    return best;
   602→  }

   606→  private onKeyDown(e: KeyboardEvent): void {
   607→    if (e.key === "Escape") {
   608→      if (this.running && !this.paused) bridge.emit(EV.pauseRequested);
   609→      return;
   610→    }
   611→    if (!this.running || this.paused) return;
   612→    let digit = -1;
   613→    if (e.key >= "0" && e.key <= "9") digit = parseInt(e.key, 10);
   614→    else if (e.code && e.code.startsWith("Numpad") && e.code.length === 7) {
   615→      const d = parseInt(e.code.slice(6), 10);
   616→      if (!Number.isNaN(d)) digit = d;
   617→    }
   618→    if (digit >= 0 && digit <= 9) this.handleCast({ digit }, "keys");
   619→  }

   621→  /** A cast arrives from the Answer Pad (source="pad") or keyboard (source="keys"). */
   622→  private handleCast(payload: CastPayload, source: Mode): void {
   623→    if (!this.running || this.paused) return;
   624→    if (this.time.now < this.inputLockUntil) return;
   625→    const digit = payload?.digit;
   626→    if (digit == null || digit < 0 || digit > 9) return;
   627→
   628→    if (source === "keys") this.keyboardCasts += 1;
   629→    else this.padCasts += 1;
   630→
   631→    let target: Wisp | null = null;
   632→    for (const w of this.wisps) {
   633→      // Skip enemies in death/attack state — they can't be hit again
   634→      if (w.animState === "death" || w.animState === "attack") continue;
   635→      if (w.answer === digit) {
   636→        if (!target || w.container.y > target.container.y) target = w;
   637→      }
   638→    }
   639→    if (target) this.correctKill(target, digit);
   640→    else this.wrongCast();
   641→  }

   643→  private correctKill(w: Wisp, digit: number): void {
   644→    SFX.play("cast_digit", digit);
   645→    const x = w.container.x;
   646→    const y = w.container.y;
   647→
   648→    // Switch to death animation state — enemy is no longer targetable
   649→    w.animState = "death";
   650→    w.animDone = false;
   651→
   652→    // Hide the question text immediately
   653→    this.tweens.add({
   654→      targets: w.questionText,
   655→      alpha: 0,
   656→      duration: TUNING.POP_DURATION,
   657→    });
   658→
   659→    // Play death animation; when done, destroy the container
   660→    const deathAnimKey = `${w.sprite.texture.key}_death`;
   661→    if (this.anims.exists(deathAnimKey)) {
   662→      this.playWispAnim(w, "death");
   663→      // Check each frame if death is done (animDone set by completion callback)
   664→      const checkDeath = () => {
   665→        if (w.animDone) {
   666→          w.container.destroy();
   667→        } else {
   668→          this.time.delayedCall(50, checkDeath);
   669→        }
   670→      };
   671→      this.time.delayedCall(100, checkDeath);
   672→    } else {
   673→      // Fallback: procedural pop + fade (original behavior)
   674→      const idx = this.wisps.indexOf(w);
   675→      if (idx >= 0) this.wisps.splice(idx, 1);
   676→      this.tweens.add({
   677→        targets: w.sprite,
   678→        scaleX: TUNING.WISP_SCALE * TUNING.POP_SCALE,
   679→        scaleY: TUNING.WISP_SCALE * TUNING.POP_SCALE,
   680→        alpha: 0,
   681→        duration: TUNING.POP_DURATION,
   682→        ease: "Quad.out",
   683→        onComplete: () => w.container.destroy(),
   684→      });
   685→    }
   686→
   687→    SFX.play("kill");
   688→    // dispatch the ten-element spell effect (cosmetic only)
   689→    this.castFx(digit, x, y);
   690→    this.spawnFloatPlus(x, y);
   691→    this.playWizardCast(digit);
   692→
   693→    this.score += TUNING.SCORE_PER_CAST;
   694→    this.streak += 1;
   695→    this.runBestStreak = Math.max(this.runBestStreak, this.streak);
   696→    this.correctCount += 1;
   697→    if (this.correctCount % this.levelCfg.rampEvery === 0) {
   698→      this.spawnInterval = Math.max(
   699→        this.levelCfg.spawnFloor,
   700→        this.spawnInterval * this.levelCfg.rampFactor,
   701→      );
   702→    }
   703→    this.emitHud();
   704→  }
   705→
   706→  private wrongCast(): void {
   707→    SFX.play("fizzle");
   708→    this.inputLockUntil = this.time.now + TUNING.WRONG_LOCK;
   709→    this.streak = 0;
   710→    this.wrongCount += 1;
   711→    // emit lock to React so the Answer Pad dims for the lock duration
   712→    bridge.emit(EV.lock, { ms: TUNING.WRONG_LOCK });
   713→    this.blinkScreen(TUNING.RED_BLINK_ALPHA, TUNING.RED_BLINK_MS);
   714→    const lowest = this.getLowest();
   715→    const sx = lowest ? lowest.container.x : TUNING.WIDTH / 2;
   716→    const sy = lowest ? lowest.container.y : TUNING.WALL_CONTACT_Y - 40;
   717→    this.spawnSparkle(sx, sy, C.ember, 6, TUNING.FIZZLE_SPARKLE_LIFE_MS);
   718→
   719→    // Wizard wrong-answer reaction (shake + ember tint)
   720→    this.playWrongAnswerReaction();
   721→
   722→    // BACKLASH (veryHard only): a wrong cast deals 1 wallHP to the keep.
   723→    if (this.levelCfg.backlash) {
   724→      this.wallHP -= 1;
   725→      SFX.play("wallHit");
   726→      this.blinkScreen(TUNING.WALL_BLINK_ALPHA, TUNING.EMBER_BLINK_MS);
   727→      if (!reducedMotion()) {
   728→        this.cameras.main.shake(TUNING.SHAKE_DURATION, TUNING.SHAKE_INTENSITY);
   729→      }
   730→      this.refreshKeepFrame();
   731→      if (this.wallHP <= 0) {
   732→        this.gameover();
   733→        return;
   734→      }
   735→    }
   736→    this.emitHud();
   737→  }
   738→
   739→  private wallHit(w: Wisp): void {
   740→    const idx = this.wisps.indexOf(w);
   741→    if (idx < 0) return;
   742→    this.wisps.splice(idx, 1);
   743→    w.container.destroy();
   744→
   745→    this.wallHP -= 1;
   746→    SFX.play("wallHit");
   747→    this.blinkScreen(TUNING.WALL_BLINK_ALPHA, TUNING.EMBER_BLINK_MS);
   748→    if (!reducedMotion()) {
   749→      this.cameras.main.shake(TUNING.SHAKE_DURATION, TUNING.SHAKE_INTENSITY);
   750→    }
   751→    this.refreshKeepFrame();
   752→
   753→    // Wizard castle-hit reaction (recoil + red tint)
   754→    this.playCastleHitReaction();
   755→
   756→    if (this.wallHP <= 0) {
   757→      this.gameover();
   758→      return;
   759→    }
   760→    this.emitHud();
   761→  }

   821→  private pulseWiz(): void {
   822→    if (!this.wiz || !this.wiz.active) return;

   824→    this.time.delayedCall(TUNING.WIZ_PULSE_MS, () => {

   829→  /**
   830→   * Play the digit-specific wizard cast animation.
   831→   *
   832→   * Validates digit is 0-9, then:
   833→   *   1. Wind-up: wizard switches to cast frame (frame 2) for SPELL_WINDUP_MS.
   834→   *   2. Release: the spell overlay PNG (spell_<digit>) appears at the staff tip.
   835→   *   3. Hold: the overlay is visible for SPELL_HOLD_MS.
   836→   *   4. Recovery: overlay fades over SPELL_FADE_MS, wizard returns to idle (frame 0).
   837→   *
   838→   * Prevents overlapping casts: if a cast is in progress, the previous spell
   839→   * overlay is destroyed before starting the new one. The wizard always
   840→   * returns to idle (frame 0) at the end.
   841→   */
   842→  private playWizardCast(digit: number): void {
   843→    if (!this.wiz || !this.wiz.active) return;
   844→    if (!Number.isInteger(digit) || digit < 0 || digit > 9) {
   845→      console.warn(`[Game] playWizardCast: invalid digit ${digit}, ignoring`);
   846→      return;
   847→    }
   848→    // Game-over has the highest priority — never start a cast during defeat
   849→    if (this.wizAnimState === "gameover") return;
   850→
   851→    // Cancel any in-progress cast overlay
   852→    if (this.activeSpellOverlay) {
   853→      this.activeSpellOverlay.destroy();
   854→      this.activeSpellOverlay = null;
   855→    }
   856→    // Cancel any in-progress reaction tween
   857→    this.cancelWizReaction();
   858→
   859→    // Wind-up: switch wizard to cast texture
   860→    this.wizAnimState = "casting";
   861→    this.setWizCastTexture();
   862→
   863→    // After wind-up, show the spell effect at the staff tip
   864→    this.time.delayedCall(TUNING.SPELL_WINDUP_MS, () => {
   865→      if (!this.wiz || !this.wiz.active) return;
   866→
   867→      const spellKey = `spell_${digit}`;
   868→      if (!this.textures.exists(spellKey)) {
   869→        console.warn(`[Game] spell texture "${spellKey}" not loaded`);
   870→        return;
   871→      }
   872→
   873→      // Create the spell overlay at the staff tip
   874→      const overlay = this.add.image(
   875→        TUNING.STAFF_TIP_X,
   876→        TUNING.STAFF_TIP_Y,
   877→        spellKey,
   878→      );
   879→      overlay.setOrigin(0.5, 0.5);
   880→      overlay.setScale(TUNING.WIZ_SCALE); // match wizard's 3x display scale
   881→      overlay.setDepth(4); // above wall (1), below wisp container (3)... actually above
   882→      overlay.setAlpha(0);
   883→      this.activeSpellOverlay = overlay;
   884→
   885→      // Quick scale-in + fade-in
   886→      this.tweens.add({
   887→        targets: overlay,
   888→        alpha: 1,
   889→        scale: TUNING.WIZ_SCALE * 1.2,
   890→        duration: 50,
   891→        ease: "Quad.out",
   892→        onComplete: () => {
   893→          // Hold phase
   894→          this.tweens.add({
   895→            targets: overlay,
   896→            scale: TUNING.WIZ_SCALE,
   897→            duration: TUNING.SPELL_HOLD_MS,
   898→            ease: "Sine.inOut",
   899→            onComplete: () => {
   900→              // Fade-out
   901→              this.tweens.add({
   902→                targets: overlay,
   903→                alpha: 0,
   904→                scale: TUNING.WIZ_SCALE * 0.5,
   905→                duration: TUNING.SPELL_FADE_MS,
   906→                ease: "Quad.in",
   907→                onComplete: () => {
   908→                  overlay.destroy();
   909→                  if (this.activeSpellOverlay === overlay) {
   910→                    this.activeSpellOverlay = null;
   911→                  }
   912→                },
   913→              });
   914→            },
   915→          });
   916→        },
   917→      });
   918→
   919→      // Recovery: wizard returns to idle after the total cast duration
   920→      this.time.delayedCall(
   921→        TUNING.SPELL_WINDUP_MS + TUNING.SPELL_HOLD_MS + TUNING.SPELL_FADE_MS,
   922→        () => {
   923→          if (this.wiz && this.wiz.active && this.wizAnimState === "casting") {
   924→            this.setWizIdleTexture();
   925→            this.resumeWizIdle();
   926→          }
   927→        },
   928→      );
   929→    });
   930→  }

   980→   * Priority: castleHit > casting. Cancels any active cast.

   982→  private playCastleHitReaction(): void {
   983→    if (!this.wiz || !this.wiz.active) return;
   984→    if (this.wizAnimState === "gameover") return; // gameover is terminal
   985→

   987→    if (this.activeSpellOverlay) {
   988→      this.activeSpellOverlay.destroy();

   990→    }
   991→    this.cancelWizReaction();
   992→
   993→    this.wizAnimState = "castleHit";
   994→    if (this.wizIdleTween) this.wizIdleTween.pause();
   995→

   997→    this.wiz.setTint(0xff6b35); // ember tint
   998→    const origX = TUNING.WIZ_X;
   999→    this.wizReactionTween = this.tweens.add({
  1000→      targets: this.wiz,
  1001→      x: origX - TUNING.WIZ_HIT_RECOIL,
  1002→      duration: TUNING.WIZ_HIT_MS / 2,
  1003→      ease: "Quad.out",
  1004→      yoyo: true,
  1005→      onComplete: () => {
  1006→        this.wiz.clearTint();
  1007→        this.wiz.setPosition(origX, TUNING.WIZ_BOTTOM_Y);
  1008→        this.resumeWizIdle();
  1009→      },
  1010→    });
  1011→  }
  1012→
  1013→  /**
  1014→   * Wrong-answer reaction: small head/body shake + ember tint.
  1015→   * ~280ms. Triggered when the player submits an incorrect answer.
  1016→   * Priority: wrongAnswer > casting (but < castleHit). Cancels active cast.
  1017→   */
  1018→  private playWrongAnswerReaction(): void {
  1019→    if (!this.wiz || !this.wiz.active) return;
  1020→    if (this.wizAnimState === "gameover") return;
  1021→    if (this.wizAnimState === "castleHit") return; // castle hit takes priority
  1022→
  1023→    // Cancel any active cast overlay
  1024→    if (this.activeSpellOverlay) {
  1025→      this.activeSpellOverlay.destroy();
  1026→      this.activeSpellOverlay = null;
  1027→    }
  1028→    this.cancelWizReaction();
  1029→
  1030→    this.wizAnimState = "wrongAnswer";
  1031→    if (this.wizIdleTween) this.wizIdleTween.pause();
  1032→
  1033→    // Shake: x oscillation + slight ember tint
  1034→    this.wiz.setTint(0xff6b35);
  1035→    const origX = TUNING.WIZ_X;
