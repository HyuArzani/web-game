    71→    71→ * INPUT MODEL:
    72→    72→ *   - The universal Answer Pad (React/DOM) is always visible during play on
    73→    73→ *     every platform. A tap emits bridge "cast" {digit}.
    74→    74→ *   - PC keyboard 0-9/numpad casts directly (desktop only). Esc pauses.

    84→    84→ * QUESTION MODEL (per-level): each wisp carries its own fixed question
    85→    85→ * generated by generateQuestion() based on the active level's operators.

   306→   306→  private refreshKeepFrame(): void {

   317→   317→  /* ----------------------------- update ------------------------------ */
   318→   318→
   319→   319→  update(time: number, deltaMs: number): void {
   320→   320→    if (!this.running || this.paused) return;
   321→   321→
   322→   322→    const cap = this.airborneCap();
   323→   323→    this.spawnTimer -= deltaMs;
   324→   324→    if (this.spawnTimer <= 0 && this.wisps.length < cap) {
   325→   325→      this.spawnWisp();
   326→   326→      this.spawnTimer += this.spawnInterval;
   327→   327→      if (this.spawnTimer < 0) this.spawnTimer = this.spawnInterval;
   328→   328→    }
   329→   329→    if (
   330→   330→      this.goSecondSpawnPending &&
   331→   331→      this.wisps.length < cap &&
   332→   332→      time >= this.runStartTime + TUNING.GO_DOUBLE_SPAWN_DELAY
   333→   333→    ) {
   334→   334→      this.goSecondSpawnPending = false;
   335→   335→      this.spawnWisp();
   336→   336→    }
   337→   337→
   338→   338→    for (const w of this.wisps) {
   339→   339→      w.container.y += w.speedPerMs * deltaMs;
   340→   340→      w.container.x =
   341→   341→        w.baseX +
   342→   342→        Math.sin(time * TUNING.SWAY_FREQ_PER_MS + w.phase) *
   343→   343→          TUNING.SWAY_AMPLITUDE;
   344→   344→      // Animation: if the enemy is in death state and animDone, destroy and remove
   345→   345→      if (w.animState === "death" && w.animDone) {
   346→   346→        const di = this.wisps.indexOf(w);
   347→   347→        if (di >= 0) this.wisps.splice(di, 1);
   348→   348→        w.container.destroy();
   349→   349→        continue;
   350→   350→      }
   351→   351→      // If in attack state and animDone, process wall hit
   352→   352→      if (w.animState === "attack" && w.animDone) {
   353→   353→        // Attack animation complete; now apply wall damage
   354→   354→        this.wallHit(w);
   355→   355→        continue;
   356→   356→      }
   357→   357→      // Skip normal movement for death/attack states
   358→   358→      if (w.animState === "death" || w.animState === "attack") continue;
   359→   359→
   360→   360→      // Normal movement (enter + idle states)
   361→   361→      w.container.y += w.speedPerMs * deltaMs;
   362→   362→      w.container.x =
   363→   363→        w.baseX +
   364→   364→        Math.sin(time * TUNING.SWAY_FREQ_PER_MS + w.phase) *
   365→   365→          TUNING.SWAY_AMPLITUDE;
   366→   366→
   367→   367→      // Frame animation: only for procedural fallback textures (no Phaser anim).
   368→   368→      // PNG spritesheets use the Phaser animation system (playWispAnim).
   369→   369→      const tex = w.sprite.texture;
   370→   370→      const animKey = `${tex.key}_idle`;
   371→   371→      if (w.animState === "idle" && !this.anims.exists(animKey) && tex && tex.frameTotal > 1) {
   372→   372→        // Procedural fallback: manual flutter
   373→   373→        w.sprite.setFrame(Math.floor(time / 140) % 4);
   374→   374→      }
   375→   375→
   376→   376→      w.trailTimer += deltaMs;
   377→   377→      if (w.trailTimer >= TUNING.WISP_TRAIL_INTERVAL_MS) {
   378→   378→        w.trailTimer = 0;
   379→   379→        this.spawnWispTrail(w);
   380→   380→      }
   381→   381→      if (w.container.y >= TUNING.WALL_CONTACT_Y) {
   382→   382→        // Switch to attack state — the wall hit logic runs when attack anim completes
   383→   383→        w.animState = "attack";
   384→   384→        w.animDone = false;
   385→   385→        this.playWispAnim(w, "attack");
   386→   386→        // If no attack animation exists (fallback), resolve immediately
   387→   387→        if (!this.anims.exists(`${w.sprite.texture.key}_attack`)) {
   388→   388→          this.wallHit(w);
   389→   389→        }
   390→   390→      }
   391→   391→    }
   392→   392→  }
   393→   393→
   394→   394→  private airborneCap(): number {
   395→   395→    const cfg = this.levelCfg;
   396→   396→    for (const phase of cfg.airborne) {
   397→   397→      if (this.correctCount < phase.until) return phase.cap;
   398→   398→    }
   399→   399→    return cfg.airborne[cfg.airborne.length - 1].cap;
   400→   400→  }

   402→   402→  /* --------------------- enemy animation system --------------------- */
   403→   403→
   404→   404→  /** Register Phaser animations for each enemy type's enter/idle/attack/death states. */
   405→   405→  private registerEnemyAnimations(): void {
   406→   406→    for (const key of ENEMY_KEYS) {
   407→   407→      const ranges = ENEMY_FRAME_RANGES[key];
   408→   408→      if (!ranges) continue;
   409→   409→      // Only register if the texture exists and has enough frames
   410→   410→      if (!this.textures.exists(key)) continue;
   411→   411→      const tex = this.textures.get(key);
   412→   412→      const totalFrames = tex.frameTotal - 1; // subtract __BASE
   413→   413→      if (totalFrames < ranges[7] + 1) continue; // not enough frames
   414→   414→
   415→   415→      const [eS, eE, iS, iE, aS, aE, dS, dE] = ranges;
   416→   416→      // Enter: play once, then transition to idle
   417→   417→      this.anims.create({
   418→   418→        key: `${key}_enter`,
   419→   419→        frames: this.anims.generateFrameNumbers(key, { start: eS, end: eE }),
   420→   420→        frameRate: 10,
   421→   421→        repeat: 0,
   422→   422→      });
   423→   423→      // Idle: loop
   424→   424→      this.anims.create({
   425→   425→        key: `${key}_idle`,

   447→   447→  /** Start an animation on a wisp sprite, handling fallback to procedural textures. */
   448→   448→  private playWispAnim(w: Wisp, state: "enter" | "idle" | "attack" | "death"): void {
   449→   449→    const animKey = `${w.sprite.texture.key}_${state}`;
   450→   450→    if (this.anims.exists(animKey)) {
   451→   451→      // Phaser Image doesn't have .play() — use the Sprite-based approach
   452→   452→      // by stopping any current anim and using the texture's frames directly
   453→   453→      // via the animation manager
   454→   454→      const anim = this.anims.get(animKey);
   455→   455→      if (anim) {
   456→   456→        // Manual frame stepping for Image objects (which don't have .play())
   457→   457→        let frameIdx = anim.frames[0].frame;
   458→   458→        w.sprite.setFrame(frameIdx);
   459→   459→        // Set up a timer-based frame cycle
   460→   460→        const frames = anim.frames;
   461→   461→        let currentFrame = 0;
   462→   462→        const frameRate = anim.frameRate || 4;
   463→   463→        const frameMs = 1000 / frameRate;
   464→   464→        // Clear any existing timer for this wisp
   465→   465→        // Use a simple recursive delayedCall approach
   466→   466→        const stepFrame = () => {
   467→   467→          if (w.animState !== state) return; // state changed, stop stepping
   468→   468→          currentFrame = (currentFrame + 1) % frames.length;
   469→   469→          if (currentFrame === 0 && anim.repeat === 0) {
   470→   470→            // Non-repeating animation completed
   471→   471→            if (state === "enter") {
   472→   472→              w.animState = "idle";
   473→   473→              this.playWispAnim(w, "idle");
   474→   474→            } else if (state === "death") {
   475→   475→              w.animDone = true;
   476→   476→            } else if (state === "attack") {
   477→   477→              w.animDone = true;
   478→   478→            }
   479→   479→            return;
   480→   480→          }
   481→   481→          w.sprite.setFrame(frames[currentFrame].frame);
   482→   482→          this.time.delayedCall(frameMs, stepFrame);
   483→   483→        };
   484→   484→        this.time.delayedCall(frameMs, stepFrame);
   485→   485→      }
   486→   486→    } else {
   487→   487→      // Fallback: procedural textures had 4 flutter frames; just set frame 0
   488→   488→      w.sprite.setFrame(0);
   489→   489→    }
   490→   490→  }

   548→   548→  /**
   549→   549→   * Generate one question for the active level, re-rolling (max 20 tries)
   550→   550→   * until its answer differs from every currently airborne wisp's answer.
   551→   551→   */
   552→   552→  private generateQuestion(): { question: string; answer: number } {
   553→   553→    const cfg = this.levelCfg;
   554→   554→    const airborne = new Set(this.wisps.map((w) => w.answer));
   555→   555→    for (let i = 0; i < TUNING.DIGIT_PICK_TRIES; i++) {
   556→   556→      const q = this.generateOneQuestion(cfg.ops);
   557→   557→      if (!airborne.has(q.answer)) return q;
   558→   558→    }
   559→   559→    return this.generateOneQuestion(cfg.ops);
   560→   560→  }
   561→   561→
   562→   562→  private generateOneQuestion(ops: Op[]): { question: string; answer: number } {
   563→   563→    const op = ops[Phaser.Math.Between(0, ops.length - 1)];
   564→   564→    const D = () => Phaser.Math.Between(TUNING.DIGIT_MIN, TUNING.DIGIT_MAX);
   565→   565→
   566→   566→    if (op === "+") {
   567→   567→      if (this.level === "veryEasy") {
   568→   568→        let a = D(), b = D();
   569→   569→        for (let i = 0; i < 20 && a + b > 9; i++) { a = D(); b = D(); }
   570→   570→        if (a + b > 9) { a = Math.min(a, 4); b = Math.min(b, 9 - a); }
   571→   571→        return { question: `${a} + ${b}`, answer: a + b };
   572→   572→      }
   573→   573→      const a = D(), b = D();
   574→   574→      return { question: `${a} + ${b}`, answer: (a + b) % 10 };
   575→   575→    }
   576→   576→
   577→   577→    if (op === "-") {
   578→   578→      let a = D(), b = D();
   579→   579→      if (a < b) { const t = a; a = b; b = t; }
   580→   580→      return { question: `${a} - ${b}`, answer: a - b };
   581→   581→    }
   582→   582→
   583→   583→    if (op === "x") {
   584→   584→      const a = D(), b = D();
   585→   585→      return { question: `${a} x ${b}`, answer: (a * b) % 10 };
   586→   586→    }
   587→   587→
   588→   588→    // divide: pick b in 2-9, q in 1-9, a = b*q. Answer = q (last digit).
   589→   589→    const b = Phaser.Math.Between(2, 9);
   590→   590→    const q = Phaser.Math.Between(1, 9);
   591→   591→    const a = b * q;
   592→   592→    return { question: `${a} / ${b}`, answer: q % 10 };
   593→   593→  }

   595→   595→  private getLowest(): Wisp | null {
   596→   596→    if (this.wisps.length === 0) return null;
   597→   597→    let best = this.wisps[0];
   598→   598→    for (const w of this.wisps) {
   599→   599→      if (w.container.y > best.container.y) best = w;
   600→   600→    }
   601→   601→    return best;
   602→   602→  }

   606→   606→  private onKeyDown(e: KeyboardEvent): void {
   607→   607→    if (e.key === "Escape") {
   608→   608→      if (this.running && !this.paused) bridge.emit(EV.pauseRequested);
   609→   609→      return;
   610→   610→    }
   611→   611→    if (!this.running || this.paused) return;
   612→   612→    let digit = -1;
   613→   613→    if (e.key >= "0" && e.key <= "9") digit = parseInt(e.key, 10);
   614→   614→    else if (e.code && e.code.startsWith("Numpad") && e.code.length === 7) {
   615→   615→      const d = parseInt(e.code.slice(6), 10);
   616→   616→      if (!Number.isNaN(d)) digit = d;
   617→   617→    }
   618→   618→    if (digit >= 0 && digit <= 9) this.handleCast({ digit }, "keys");
   619→   619→  }

   621→   621→  /** A cast arrives from the Answer Pad (source="pad") or keyboard (source="keys"). */
   622→   622→  private handleCast(payload: CastPayload, source: Mode): void {
   623→   623→    if (!this.running || this.paused) return;
   624→   624→    if (this.time.now < this.inputLockUntil) return;
   625→   625→    const digit = payload?.digit;
   626→   626→    if (digit == null || digit < 0 || digit > 9) return;
   627→   627→
   628→   628→    if (source === "keys") this.keyboardCasts += 1;
   629→   629→    else this.padCasts += 1;
   630→   630→
   631→   631→    let target: Wisp | null = null;
   632→   632→    for (const w of this.wisps) {
   633→   633→      // Skip enemies in death/attack state — they can't be hit again
   634→   634→      if (w.animState === "death" || w.animState === "attack") continue;
   635→   635→      if (w.answer === digit) {
   636→   636→        if (!target || w.container.y > target.container.y) target = w;
   637→   637→      }
   638→   638→    }
   639→   639→    if (target) this.correctKill(target, digit);
   640→   640→    else this.wrongCast();
   641→   641→  }

   643→   643→  private correctKill(w: Wisp, digit: number): void {
   644→   644→    SFX.play("cast_digit", digit);
   645→   645→    const x = w.container.x;
   646→   646→    const y = w.container.y;
   647→   647→
   648→   648→    // Switch to death animation state — enemy is no longer targetable
   649→   649→    w.animState = "death";
   650→   650→    w.animDone = false;
   651→   651→
   652→   652→    // Hide the question text immediately
   653→   653→    this.tweens.add({
   654→   654→      targets: w.questionText,
   655→   655→      alpha: 0,
   656→   656→      duration: TUNING.POP_DURATION,
   657→   657→    });
   658→   658→
   659→   659→    // Play death animation; when done, destroy the container
   660→   660→    const deathAnimKey = `${w.sprite.texture.key}_death`;
   661→   661→    if (this.anims.exists(deathAnimKey)) {
   662→   662→      this.playWispAnim(w, "death");
   663→   663→      // Check each frame if death is done (animDone set by completion callback)
   664→   664→      const checkDeath = () => {
   665→   665→        if (w.animDone) {
   666→   666→          w.container.destroy();
   667→   667→        } else {
   668→   668→          this.time.delayedCall(50, checkDeath);
   669→   669→        }
   670→   670→      };
   671→   671→      this.time.delayedCall(100, checkDeath);
   672→   672→    } else {
   673→   673→      // Fallback: procedural pop + fade (original behavior)
   674→   674→      const idx = this.wisps.indexOf(w);
   675→   675→      if (idx >= 0) this.wisps.splice(idx, 1);
   676→   676→      this.tweens.add({
   677→   677→        targets: w.sprite,
   678→   678→        scaleX: TUNING.WISP_SCALE * TUNING.POP_SCALE,
   679→   679→        scaleY: TUNING.WISP_SCALE * TUNING.POP_SCALE,
   680→   680→        alpha: 0,
   681→   681→        duration: TUNING.POP_DURATION,
   682→   682→        ease: "Quad.out",
   683→   683→        onComplete: () => w.container.destroy(),
   684→   684→      });
   685→   685→    }
   686→   686→
   687→   687→    SFX.play("kill");
   688→   688→    // dispatch the ten-element spell effect (cosmetic only)
   689→   689→    this.castFx(digit, x, y);
   690→   690→    this.spawnFloatPlus(x, y);
   691→   691→    this.playWizardCast(digit);
   692→   692→
   693→   693→    this.score += TUNING.SCORE_PER_CAST;
   694→   694→    this.streak += 1;
   695→   695→    this.runBestStreak = Math.max(this.runBestStreak, this.streak);
   696→   696→    this.correctCount += 1;
   697→   697→    if (this.correctCount % this.levelCfg.rampEvery === 0) {
   698→   698→      this.spawnInterval = Math.max(
   699→   699→        this.levelCfg.spawnFloor,
   700→   700→        this.spawnInterval * this.levelCfg.rampFactor,
   701→   701→      );
   702→   702→    }
   703→   703→    this.emitHud();
   704→   704→  }

   829→   829→  /**
   830→   830→   * Play the digit-specific wizard cast animation.
   831→   831→   *
   832→   832→   * Validates digit is 0-9, then:
   833→   833→   *   1. Wind-up: wizard switches to cast frame (frame 2) for SPELL_WINDUP_MS.
   834→   834→   *   2. Release: the spell overlay PNG (spell_<digit>) appears at the staff tip.
   835→   835→   *   3. Hold: the overlay is visible for SPELL_HOLD_MS.
   836→   836→   *   4. Recovery: overlay fades over SPELL_FADE_MS, wizard returns to idle (frame 0).
   837→   837→   *
   838→   838→   * Prevents overlapping casts: if a cast is in progress, the previous spell
   839→   839→   * overlay is destroyed before starting the new one. The wizard always
   840→   840→   * returns to idle (frame 0) at the end.
   841→   841→   */
   842→   842→  private playWizardCast(digit: number): void {
   843→   843→    if (!this.wiz || !this.wiz.active) return;
   844→   844→    if (!Number.isInteger(digit) || digit < 0 || digit > 9) {
   845→   845→      console.warn(`[Game] playWizardCast: invalid digit ${digit}, ignoring`);
   846→   846→      return;
   847→   847→    }
   848→   848→    // Game-over has the highest priority — never start a cast during defeat
   849→   849→    if (this.wizAnimState === "gameover") return;
   850→   850→
   851→   851→    // Cancel any in-progress cast overlay
   852→   852→    if (this.activeSpellOverlay) {
   853→   853→      this.activeSpellOverlay.destroy();
   854→   854→      this.activeSpellOverlay = null;
   855→   855→    }
   856→   856→    // Cancel any in-progress reaction tween
   857→   857→    this.cancelWizReaction();
   858→   858→
   859→   859→    // Wind-up: switch wizard to cast texture
   860→   860→    this.wizAnimState = "casting";
   861→   861→    this.setWizCastTexture();
   862→   862→
   863→   863→    // After wind-up, show the spell effect at the staff tip
   864→   864→    this.time.delayedCall(TUNING.SPELL_WINDUP_MS, () => {
